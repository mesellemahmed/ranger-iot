    14	def _capsule() -> RekeyCapsule:
    15	    return RekeyCapsule(
    16	        capsule_id="g1-e8",
    17	        group_id=1,
    18	        epoch=8,
    19	        logical_ciphertexts=3,
    20	        payload_bytes=301,
    21	    )

    24	def _fragments() -> tuple[CodedFragment, ...]:
    25	    return StaticErasureCoder.encode(_capsule(), CodingParameters(3, 5))

    28	def test_coding_parameters_report_redundancy_and_fragment_size() -> None:
    29	    parameters = CodingParameters(data_fragments=3, total_fragments=5)
    30

    32	    assert parameters.fragment_payload_bytes(301) == 101
    33	    assert parameters.fragment_payload_bytes(300) == 100

    53	def test_static_coder_emits_deterministic_k_of_n_metadata() -> None:
    54	    fragments = _fragments()
    55	
    56	    assert len(fragments) == 5
    57	    assert tuple(fragment.fragment_index for fragment in fragments) == (0, 1, 2, 3, 4)
    58	    assert {fragment.payload_bytes for fragment in fragments} == {101}
    59	    assert {fragment.data_fragments for fragment in fragments} == {3}
    60	    assert {fragment.total_fragments for fragment in fragments} == {5}
    61	    assert {fragment.epoch for fragment in fragments} == {8}

    64	def test_accumulator_decodes_any_three_distinct_fragments() -> None:
    65	    fragments = _fragments()
    66	    accumulator = FragmentAccumulator("g1-e8", 1, 8, CodingParameters(3, 5))
    67	
    68	    assert accumulator.accept(fragments[0])
    69	    assert accumulator.accept(fragments[3])
    70	    assert not accumulator.decodable
    71	    assert accumulator.remaining_needed == 1
    72	
    73	    assert accumulator.accept(fragments[4])
    74	    assert accumulator.decodable
    75	    assert accumulator.unique_received == 3
    76	    assert accumulator.remaining_needed == 0
    77

    79	def test_accumulator_does_not_count_duplicate_fragment_twice() -> None:
    80	    fragments = _fragments()
    81	    accumulator = FragmentAccumulator("g1-e8", 1, 8, CodingParameters(3, 5))
    82	
    83	    assert accumulator.accept(fragments[0])
    84	    assert not accumulator.accept(fragments[0])
    85	    assert accumulator.accept(fragments[1])
    86	
    87	    assert accumulator.unique_received == 2
    88	    assert not accumulator.decodable
