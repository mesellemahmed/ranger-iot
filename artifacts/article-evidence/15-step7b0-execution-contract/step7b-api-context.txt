from dataclasses import dataclass

from ranger_iot.group import GroupState
from ranger_iot.protocols.lkh import LkhKeyTree

from ranger_iot.sim.core import SimulationEngine

@dataclass(frozen=True, slots=True)
class _DropPacketIds:
    packet_ids: frozenset[int]
    sender: int = 0
    receiver: int = 1

    def should_drop(
        self,
        *,
        sender: int,
        receiver: int,
        packet: Packet,
        rng: RandomStream,
    ) -> bool:
        del rng
        return (
            sender == self.sender
            and receiver == self.receiver
            and packet.packet_id in self.packet_ids
        )


def _build_protocol(
    *,
    loss_model: LossModel | None = None,
    revoked_member: int = 4,
    coding: CodingParameters | None = None,
) -> tuple[SimulationEngine, GroupState, NetworkMetrics, RangerStaticProtocol]:
    engine = SimulationEngine()
    node_ids = (0, 1, 2, 3, 4)
    nodes = NodeRegistry(node_ids)
    topology = StaticTopology.from_edges(
        nodes=node_ids,
        edges=((0, 1), (1, 2), (2, 3), (3, 4)),
    )
    metrics = NetworkMetrics()
    network = Network(
        engine=engine,
        topology=topology,
        nodes=nodes,
        loss_model=loss_model or NoLoss(),
        delay_model=ConstantDelay(10),
        rng=RandomStream(42, "ranger-static-test"),
        observers=(metrics,),
    )
    transport = FloodTransport(
        network=network,
        topology=topology,
        packet_factory=PacketFactory(),
    )
    group = GroupState(group_id=7, manager_id=0, member_ids={1, 2, 3, 4})
    tree = LkhKeyTree({1, 2, 3, 4})
    protocol = RangerStaticProtocol(
        group=group,
        tree=tree,
        transport=transport,
        parameters=RangerStaticParameters(
            coding=coding or CodingParameters(data_fragments=3, total_fragments=5),
            ciphertext_payload_bytes=64,
        ),
    )
    assert revoked_member in group.member_ids
    return engine, group, metrics, protocol


def test_static_ranger_completes_without_loss() -> None:
    engine, group, metrics, protocol = _build_protocol()

    session = protocol.start_rekey(revoked_nodes={4})
    processed = engine.run()
    stats = protocol.transport_stats(session.epoch)
    snapshot = metrics.snapshot()

    assert processed == 20
    assert session.completed
    assert session.accepted_at_us == {1: 10, 2: 20, 3: 30}
    assert group.member_epoch(1) == 1
    assert group.member_epoch(2) == 1
    assert group.member_epoch(3) == 1
    assert group.member_epoch(4) == 0
    assert not group.is_authorized(4)

    assert stats.logical_ciphertexts == 3
    assert stats.capsule_payload_bytes == 192
    assert stats.data_fragments == 3
    assert stats.total_fragments == 5
    assert stats.fragment_payload_bytes == 64
    assert stats.flood_injections == 5
    assert snapshot.transmissions == 20
    assert snapshot.bytes_sent == 1280
    assert snapshot.drops == 0


def test_static_ranger_tolerates_two_lost_origin_fragments_for_three_of_five() -> None:

def test_static_ranger_rejects_group_tree_mismatch() -> None:
    engine = SimulationEngine()
    nodes = NodeRegistry((0, 1, 2, 3, 4))
    topology = StaticTopology.from_edges(
        nodes=(0, 1, 2, 3, 4),
        edges=((0, 1), (1, 2), (2, 3), (3, 4)),
    )
    network = Network(
        engine=engine,
        topology=topology,
        nodes=nodes,
        loss_model=NoLoss(),
        delay_model=ConstantDelay(10),
        rng=RandomStream(42, "mismatch"),
    )
    transport = FloodTransport(
        network=network,
        topology=topology,
        packet_factory=PacketFactory(),
    )
    group = GroupState(group_id=7, manager_id=0, member_ids={1, 2, 3, 4})
    tree = LkhKeyTree({1, 2, 3})

    try:
        RangerStaticProtocol(group=group, tree=tree, transport=transport)
    except ValueError as exc:
        assert str(exc) == "group and LKH tree must contain the same members"
    else:
        raise AssertionError("expected group/tree mismatch to be rejected")

    assert stats.peer_recovery_fragment_attempts == 2

def test_v2_rejects_invalid_forwarded_feedback_and_unknown_epoch() -> None:
    engine, _, _, protocol = _build_protocol()
    session = protocol.start_rekey(revoked_nodes={4})
    engine.run()

    invalid = FloodMessage(
        message_id=999,
        origin=1,
        kind=protocol.ADAPTIVE_NACK_KIND,
        payload_bytes=16,
        created_at_us=engine.now_us,
        epoch=session.epoch,
        payload=object(),
    )
    with pytest.raises(TypeError, match="invalid RANGER adaptive-NACK"):
        protocol._handle_adaptive_nack(node_id=1, message=invalid)

    request = AdaptiveRedundancyRequest(
        capsule_id="capsule",
        group_id=9,
        epoch=session.epoch,
        requester_id=1,
        round_id=1,
        unique_received=2,
    )
    mismatch = FloodMessage(
        message_id=1000,
        origin=1,
        kind=protocol.ADAPTIVE_NACK_KIND,
        payload_bytes=16,
        created_at_us=engine.now_us,
        epoch=session.epoch + 1,
        payload=request,
    )
    with pytest.raises(ValueError, match="epoch mismatch"):
        protocol._handle_adaptive_nack(node_id=1, message=mismatch)

    with pytest.raises(KeyError, match="unknown adaptive-v2 epoch"):
        protocol._require_suppression(999)

import pytest

from ranger_iot.sim.routing import shortest_path
from ranger_iot.sim.topology_factory import build_topology


def test_line_topology_places_root_at_one_end() -> None:
    topology = build_topology("line", nodes=(7, 1, 2, 3), root_id=7)

    assert topology.nodes == (1, 2, 3, 7)
    assert shortest_path(topology, 7, 3) == (7, 1, 2, 3)


def test_star_topology_connects_every_node_directly_to_root() -> None:
    topology = build_topology("star", nodes=(0, 1, 2, 3, 4), root_id=0)

    assert topology.neighbors(0) == (1, 2, 3, 4)
    assert shortest_path(topology, 0, 4) == (0, 4)
    assert shortest_path(topology, 1, 4) == (1, 0, 4)


def test_balanced_tree_is_breadth_first_and_deterministic() -> None:
    topology = build_topology("balanced-tree", nodes=range(7), root_id=0)

    assert topology.neighbors(0) == (1, 2)
    assert topology.neighbors(1) == (0, 3, 4)
    assert topology.neighbors(2) == (0, 5, 6)
    assert shortest_path(topology, 0, 6) == (0, 2, 6)

    topology = build_topology("grid", nodes=range(9), root_id=0)

    assert topology.neighbors(0) == (1, 3)
    assert topology.neighbors(4) == (1, 3, 5, 7)
    assert topology.neighbors(8) == (5, 7)
    assert shortest_path(topology, 0, 8) == (0, 1, 2, 5, 8)

    topology = build_topology("grid", nodes=range(5), root_id=0)

import pytest

from ranger_iot.sim.core import ScheduledEvent, SimulationEngine
from ranger_iot.sim.delay import ConstantDelay
from ranger_iot.sim.loss import BernoulliLoss, NoLoss
from ranger_iot.sim.metrics import DropReason, NetworkMetrics
from ranger_iot.sim.network import Network
from ranger_iot.sim.node import NodeRegistry
from ranger_iot.sim.packet import PacketFactory
from ranger_iot.sim.randomness import RandomStream
from ranger_iot.sim.topology import StaticTopology


def build_network(
    *, loss_probability: float = 0.0
) -> tuple[
    SimulationEngine,
    NodeRegistry,
    NetworkMetrics,
    Network,
    PacketFactory,
]:
    engine = SimulationEngine()
    nodes = NodeRegistry([0, 1, 2])
    topology = StaticTopology.from_edges(nodes=[0, 1, 2], edges=[(0, 1), (0, 2)])
    metrics = NetworkMetrics()
    network = Network(
        engine=engine,
        topology=topology,
        nodes=nodes,
        loss_model=BernoulliLoss(loss_probability),
        delay_model=ConstantDelay(10),
        rng=RandomStream(123, "network"),
        observers=[metrics],
    )
    return engine, nodes, metrics, network, PacketFactory()


def test_unicast_delivery_and_metrics() -> None:
    engine, _, metrics, network, factory = build_network()
    received: list[tuple[int, int, int]] = []
    network.register_receiver(
        1,
        lambda packet, sender, at_us: received.append((packet.packet_id, sender, at_us)),
    )
    packet = factory.create(source=0, kind="hello", payload_bytes=40, created_at_us=0)

    assert network.send(sender=0, receiver=1, packet=packet)
    assert engine.pending_events == 1
    assert engine.run() == 1
    assert received == [(0, 0, 10)]

    snapshot = metrics.snapshot()
    assert snapshot.transmissions == 1
    assert snapshot.deliveries == 1
    assert snapshot.drops == 0
    assert snapshot.bytes_sent == 40
    assert snapshot.bytes_delivered == 40


def test_broadcast_uses_deterministic_neighbor_order() -> None:
    engine, _, metrics, network, factory = build_network()
    observed: list[int] = []
    network.register_receiver(1, lambda packet, sender, at_us: observed.append(1))
    network.register_receiver(2, lambda packet, sender, at_us: observed.append(2))
    packet = factory.create(source=0, kind="broadcast", payload_bytes=8, created_at_us=0)

    assert network.broadcast(sender=0, packet=packet) == 2
    assert engine.run() == 2
    assert observed == [1, 2]
    assert metrics.snapshot().transmissions_by_node == {0: 2}


def test_loss_and_inactive_receiver_are_recorded() -> None:
    engine, nodes, metrics, network, factory = build_network(loss_probability=1.0)
    packet = factory.create(source=0, kind="x", payload_bytes=5, created_at_us=0)

    snapshot = metrics.snapshot()
    assert snapshot.transmissions == 2
    assert snapshot.drops == 2
    assert snapshot.drops_by_reason == {
        DropReason.LOSS_MODEL: 1,
        DropReason.RECEIVER_INACTIVE: 1,
    }

def test_network_send_validation() -> None:
    _, nodes, _, network, factory = build_network()
    packet = factory.create(source=0, kind="x", payload_bytes=1, created_at_us=0)

    with pytest.raises(KeyError, match="unknown sender"):
        network.send(sender=9, receiver=1, packet=packet)
    with pytest.raises(KeyError, match="unknown receiver"):
        network.send(sender=0, receiver=9, packet=packet)

    nodes.set_active(0, False)
    with pytest.raises(RuntimeError, match="inactive"):
        network.send(sender=0, receiver=1, packet=packet)
    nodes.set_active(0, True)

    with pytest.raises(ValueError, match="must differ"):
        network.send(sender=0, receiver=0, packet=packet)

    wrong_source = factory.create(source=1, kind="x", payload_bytes=1, created_at_us=0)
    with pytest.raises(ValueError, match="source does not match"):
        network.send(sender=0, receiver=1, packet=wrong_source)

    with pytest.raises(ValueError, match="not directly connected"):
        network.send(sender=1, receiver=2, packet=wrong_source)

            delay_model=ConstantDelay(0),
            rng=RandomStream(1),
        )

    assert engine.pending_events == 0

import pytest

from ranger_iot.sim.randomness import RandomStream


def test_random_stream_replays_exact_sequence() -> None:
    left = RandomStream(42, "loss")
    right = RandomStream(42, "loss")

    assert left.derived_seed == right.derived_seed
    assert [left.random() for _ in range(5)] == [right.random() for _ in range(5)]


def test_forked_streams_are_namespace_isolated() -> None:
    root = RandomStream(7)
    loss = root.fork("loss")
    delay = root.fork("delay")
    loss_replay = RandomStream(7).fork("loss")

    assert loss.namespace == "root/loss"
    assert loss.derived_seed != delay.derived_seed
    assert [loss.random() for _ in range(3)] == [loss_replay.random() for _ in range(3)]


def test_random_stream_helpers_and_validation() -> None:
    rng = RandomStream(3)

    assert 1.0 <= rng.uniform(1.0, 2.0) <= 2.0
    assert 2 <= rng.integer(2, 5) < 5
    assert rng.master_seed == 3

    with pytest.raises(ValueError, match="master_seed"):
        RandomStream(-1)
    with pytest.raises(ValueError, match="namespace"):
        RandomStream(1, "")
    with pytest.raises(ValueError, match="namespace"):
        rng.fork("")
    with pytest.raises(ValueError, match="greater"):
        rng.uniform(2.0, 1.0)
    with pytest.raises(ValueError, match="greater"):
        rng.integer(3, 3)
